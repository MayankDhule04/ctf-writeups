# Step 10: Try connecting to the registry with Python and list repos
python3 << 'PYEND'
import urllib.request
import json

REG = "http://172.27.0.1:5000"

try:
    # Try to list catalog
    print("\n=== Catalog ===")
    r = urllib.request.Request(REG + "/v2/_catalog")
    data = json.load(urllib.request.urlopen(r, timeout=10))
    print(json.dumps(data, indent=2))
    
    # Try to get tags for each repo
    for repo in data.get("repositories", []):
        print(f"\n=== Tags for {repo} ===")
        try:
            r = urllib.request.Request(REG + f"/v2/{repo}/tags/list")
            tags = json.load(urllib.request.urlopen(r, timeout=10))
            print(json.dumps(tags, indent=2))
        except Exception as e:
            print(f"Error: {e}")

except Exception as e:
    print(f"Registry error: {e}")
    print("\nRegistry might not be accessible or doesn't exist")
PYEND
